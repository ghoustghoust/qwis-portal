### 1. 后端修复：增强 refresh-all 接口支持 type 过滤

**文件**: `server/routes/sources.js:112-139`

**修复前** (L116):
```javascript
router.post('/refresh-all', async (req, res) => {
  const skipBreaker = req.query.skipBreaker === '1';
  const rows = db.prepare('SELECT * FROM sources WHERE enabled=1 ORDER BY id').all(); // ❌ 无类型过滤
  // ...
});
```

**修复后**:
```javascript
router.post('/refresh-all', async (req, res) => {
  const skipBreaker = req.query.skipBreaker === '1';
  const filterType = req.query.type; // ← 新增类型过滤参数
  
  // 动态构建 SQL 查询条件
  let conds = ['enabled=1'];
  let args = [];
  if (filterType) {
    conds.push('type=?');
    args.push(filterType);
  }
  const where = conds.join(' AND ');
  
  const rows = db.prepare(`SELECT * FROM sources WHERE ${where} ORDER BY id`).all(...args); // ✅ 带类型过滤
  // ...
});
```

**修改行数**: +13 行 -2 行

    loadSources();

**批量解冻函数**:
```javascript
// F32c: 批量解冻已熔断的 B 站订阅
const unfreezeAll = async () => {
  const frozenBili = paused.filter((s) => s.type === 'bilibili');
  if (!frozenBili.length) {
    toast('没有已熔断的 B 站订阅');
    return;
  }
  if (!window.confirm(`确定批量解冻 ${frozenBili.length} 个熔断的 B 站订阅？`)) return;
  setBusy('unfreeze');
  try {
    let restored = 0;
    for (const s of frozenBili) {
      await api.put(`/api/sources/${s.id}/toggle`).catch(() => {});
      restored++;
    }
    toast(`已恢复 ${restored} 个 B 站订阅`);
    loadSources();
    reloadStatus();
  } catch (e) {
    toast(e.message || '解冻失败');
  } finally {
    setBusy('');
  }
};
```
